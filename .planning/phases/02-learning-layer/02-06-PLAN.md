```python
@router.get("/assignments/{assignment_id}/leaderboard")
async def get_leaderboard(
    assignment_id: str,
    current_user: User = Depends(get_current_user),
    db: AsyncSession = Depends(get_db),
):
    # Fetch assignment, verify it's open (or caller is instructor)
    # Query: best submission per user (max score) ordered by score DESC, WNS numeric DESC
    # For best-per-user: subquery to find max score per user, then join back
    # Anonymize: add is_self flag, never return names in response
    # Build ranked response list
    from sqlalchemy import text, func

    # Best submission per user subquery
    best_score_sub = (
        select(
            Submission.user_id,
            func.max(Submission.score).label("best_score"),
        )
        .where(Submission.assignment_id == assignment_id)
        .group_by(Submission.user_id)
        .subquery()
    )

    stmt = (
        select(best_score_sub.c.user_id, best_score_sub.c.best_score,
               text("(runs.ppa->>'worst_negative_slack')::numeric as wns"))
        .select_from(best_score_sub)
        .join(Submission, (Submission.user_id == best_score_sub.c.user_id) &
              (Submission.assignment_id == assignment_id) &
              (Submission.score == best_score_sub.c.best_score))
        .join(Run, Submission.run_id == Run.id)
        .order_by(
            best_score_sub.c.best_score.desc().nullslast(),
            text("(runs.ppa->>'worst_negative_slack')::numeric DESC NULLS LAST"),
        )
    )

    results = await db.execute(stmt)
    rows = results.all()

    return [
        {
            "rank": i + 1,
            "score": row.best_score,
            "wns": row.wns,
            "user_id": str(row.user_id),
            "is_self": str(row.user_id) == str(current_user.id),
        }
        for i, row in enumerate(rows)
    ]
```

```tsx
// Props: runs: Array<{id: string; created_at: string; ppa: Record<string, number>; config: Record<string, string>}>
// Metrics to compare: WNS, TNS, DRC violations, core utilization, total power
// Color coding: compute min/max across selected runs per metric; best=green, worst=red, middle=yellow
// Config diff: show only config keys where value differs across runs
// Max 4 runs (per CONTEXT.md decisions)

const METRICS = [
  { key: "worst_negative_slack", label: "WNS (ns)", higherBetter: true },
  { key: "total_negative_slack", label: "TNS (ns)", higherBetter: true },
  { key: "drc_violations", label: "DRC Violations", higherBetter: false },
  { key: "core_utilization", label: "Core Utilization (%)", higherBetter: false },
  { key: "total_power", label: "Total Power (W)", higherBetter: false },
];

function colorClass(value: number, values: number[], higherBetter: boolean): string {
  const sorted = [...values].sort((a, b) => higherBetter ? b - a : a - b);
  if (value === sorted[0]) return "bg-green-100 text-green-800";
  if (value === sorted[sorted.length - 1]) return "bg-red-100 text-red-800";
  return "bg-yellow-50 text-yellow-800";
}
```

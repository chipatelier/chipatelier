OpenDB subprocess query script (from RESEARCH.md Pattern 6):
```python
QUERY_SCRIPT_TEMPLATE = """
import openroad
import odb
import json
import sys

db = odb.dbDatabase.create()
odb.read_db(db, "{odb_path}")
chip = db.getChip()
block = chip.getBlock()
dbu = block.getDbUnitsPerMicron()

x_dbu = int({x_um} * dbu)
y_dbu = int({y_um} * dbu)
tol_dbu = int({tolerance_um} * dbu)

results = []
for inst in block.getInsts():
    bbox = inst.getBBox()
    if (bbox.xMin() - tol_dbu <= x_dbu <= bbox.xMax() + tol_dbu and
        bbox.yMin() - tol_dbu <= y_dbu <= bbox.yMax() + tol_dbu):
        master = inst.getMaster()
        nets = [it.getNet().getName() for it in inst.getITerms() if it.getNet()]
        results.append({
            "name": inst.getName(),
            "master": master.getName() if master else None,
            "nets": nets,
        })

print(json.dumps(results))
"""
```

```python
# PNG pixel (0,0) is top-left; EDA coordinates (0,0) is bottom-left

y_um = (1 - click_y_px / png_height_px) * (bbox_ymax_um - bbox_ymin_um) + bbox_ymin_um
```

Create `backend/app/api/routes/query.py`:

```python
import json
import subprocess
import tempfile
import os
from pathlib import Path
from fastapi import APIRouter, Depends, HTTPException, Query
from sqlalchemy.ext.asyncio import AsyncSession
from sqlalchemy import select
from app.core.database import get_db
from app.core.security import get_current_user
from app.core.config import settings
from app.models.user import User
from app.models.run import Run
from app.schemas.query import InspectResponse, InspectElement

router = APIRouter()

QUERY_SCRIPT_TEMPLATE = """
import odb, json, sys
db = odb.dbDatabase.create()
odb.read_db(db, "{odb_path}")
chip = db.getChip()
block = chip.getBlock()
dbu = block.getDbUnitsPerMicron()
x_dbu = int({x_um} * dbu)
y_dbu = int({y_um} * dbu)
tol_dbu = int({tolerance_um} * dbu)
results = []
for inst in block.getInsts():
    bbox = inst.getBBox()
    if (bbox.xMin() - tol_dbu <= x_dbu <= bbox.xMax() + tol_dbu and
            bbox.yMin() - tol_dbu <= y_dbu <= bbox.yMax() + tol_dbu):
        master = inst.getMaster()
        nets = [it.getNet().getName() for it in inst.getITerms() if it.getNet()]
        results.append(dict(name=inst.getName(), master=master.getName() if master else None, nets=nets))
print(json.dumps(results))
"""

@router.get("/query/{run_id}", response_model=InspectResponse)
async def click_to_inspect(
    run_id: str,
    x_um: float = Query(...),
    y_um: float = Query(...),
    tolerance_um: float = Query(default=1.0),
    current_user: User = Depends(get_current_user),
    db: AsyncSession = Depends(get_db),
):
    # Fetch run + verify ownership
    stmt = select(Run).where(Run.id == run_id)
    result = await db.execute(stmt)
    run = result.scalar_one_or_none()
    if not run:
        raise HTTPException(404, "Run not found")

    # Ownership check: fetch project to get user_id
    from sqlalchemy.orm import selectinload
    stmt2 = select(Run).options(selectinload(Run.project)).where(Run.id == run_id)
    result2 = await db.execute(stmt2)
    run_with_project = result2.scalar_one()
    if str(run_with_project.project.user_id) != str(current_user.id):
        raise HTTPException(403, "Not your run")

    if not run.artifact_path:
        raise HTTPException(400, "Run has no artifacts — complete the run first")

    # Download ODB from MinIO to temp dir
    from app.services.storage_service import StorageService
    storage = StorageService()

    # Determine highest completed stage ODB key
    stage_odb_map = {
        "finish": "6_final.odb",
        "route":  "5_route.odb",
        "cts":    "4_cts.odb",
        "place":  "3_place.odb",
        "floorplan": "2_floorplan.odb",
        "synth":  "1_synth.odb",
    }
    stage = run.stage_completed or "route"
    odb_filename = stage_odb_map.get(stage, "5_route.odb")
    odb_minio_key = f"{run.artifact_path}/{odb_filename}"

    tmpdir = tempfile.mkdtemp(prefix="chipatelier_query_")
    odb_local_path = os.path.join(tmpdir, odb_filename)

    try:
        await storage.download_file(odb_minio_key, odb_local_path)

        script = QUERY_SCRIPT_TEMPLATE.format(
            odb_path=odb_local_path,
            x_um=x_um,
            y_um=y_um,
            tolerance_um=tolerance_um,
        )

        # Run openroad subprocess inside ORFS container image (read-only mount)
        cmd = [
            "docker", "run", "--rm",
            "--network", "none",
            "--user", "orfs:orfs",
            "--cap-drop", "ALL",
            "--security-opt", "no-new-privileges",
            "-v", f"{tmpdir}:{tmpdir}:ro",
            settings.ORFS_IMAGE,
            "openroad", "-python", "-e", script,
        ]

        proc = subprocess.run(
            cmd,
            capture_output=True,
            text=True,
            timeout=30,
        )

        if proc.returncode != 0:
            raise HTTPException(500, f"OpenROAD query failed: {proc.stderr[:500]}")

        elements_raw = json.loads(proc.stdout.strip())
        elements = [InspectElement(**e) for e in elements_raw]

        return InspectResponse(elements=elements, run_id=run_id, x_um=x_um, y_um=y_um)

    finally:
        import shutil
        shutil.rmtree(tmpdir, ignore_errors=True)
```

```typescript
import { apiClient } from "./client";  // existing axios client

export interface InspectElement {
  name: string;
  master: string | null;
  nets: string[];
}

export interface InspectResponse {
  elements: InspectElement[];
  run_id: string;
  x_um: number;
  y_um: number;
}

export async function clickToInspect(
  runId: string,
  xUm: number,
  yUm: number,
  toleranceUm: number = 1.0,
): Promise<InspectResponse> {
  const resp = await apiClient.get(`/query/${runId}`, {
    params: { x_um: xUm, y_um: yUm, tolerance_um: toleranceUm },
  });
  return resp.data;
}
```

```tsx
import React from "react";
import { InspectElement } from "../../api/query";

interface InspectSidebarProps {
  elements: InspectElement[] | null;  // null = no query yet
  isLoading: boolean;
  onDismiss: () => void;
}

export function InspectSidebar({ elements, isLoading, onDismiss }: InspectSidebarProps) {
  if (elements === null && !isLoading) return null;  // hidden when no query made

  return (
    <div className="inspect-sidebar w-80 border-l bg-white flex flex-col h-full">
      <div className="flex items-center justify-between p-3 border-b">
        <span className="font-semibold text-sm">Layout Inspector</span>
        <button onClick={onDismiss} aria-label="Close inspector" className="text-gray-400 hover:text-gray-600">
          ✕
        </button>
      </div>
      <div className="flex-1 overflow-y-auto p-3">
        {isLoading && <p className="text-sm text-gray-500">Querying...</p>}
        {!isLoading && elements !== null && elements.length === 0 && (
          <p className="text-sm text-gray-500">No element at this location</p>
        )}
        {!isLoading && elements && elements.map((el, i) => (
          <div key={i} className="mb-4 p-3 bg-gray-50 rounded border">
            <p className="font-mono text-sm font-bold">{el.name}</p>
            {el.master && <p className="text-xs text-gray-600 mt-1">Type: {el.master}</p>}
            {el.nets.length > 0 && (
              <div className="mt-1">
                <p className="text-xs text-gray-500">Nets:</p>

                  {el.nets.map(net => (
                    <span key={net} className="text-xs bg-blue-100 text-blue-800 px-1.5 py-0.5 rounded font-mono">{net}</span>
                  ))}
                </div>
              </div>

          </div>
        ))}
      </div>
    </div>
  );
}
```

```typescript
const handleImageClick = async (e: React.MouseEvent<HTMLImageElement>) => {
  if (!runId || !layoutBbox) return;
  const rect = e.currentTarget.getBoundingClientRect();
  const clickX = e.clientX - rect.left;
  const clickY = e.clientY - rect.top;
  const pxW = rect.width;
  const pxH = rect.height;
  // Map pixel → micron with Y-axis inversion (PNG Y-down, EDA Y-up)
  const xUm = (clickX / pxW) * (layoutBbox.xmax - layoutBbox.xmin) + layoutBbox.xmin;

  setInspectLoading(true);
  try {
    const result = await clickToInspect(runId, xUm, yUm);
    setInspectElements(result.elements);
  } finally {
    setInspectLoading(false);
  }
};
```

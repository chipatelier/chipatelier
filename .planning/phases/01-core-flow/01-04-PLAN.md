    **backend/app/api/websocket.py** — WebSocket log streaming endpoint:

    ```python
    from fastapi import APIRouter, WebSocket, WebSocketDisconnect, Query, HTTPException
    import redis.asyncio as aioredis
    from app.core.redis import get_redis
    from app.core.security import decode_token
    import jwt

    router = APIRouter()

    @router.websocket("/jobs/{run_id}/logs/stream")
    async def log_stream(websocket: WebSocket, run_id: str,
                         token: str = Query(...)):
        # Validate token before accepting connection
        try:
            payload = decode_token(token)
            if payload.get("type") != "access":
                await websocket.close(code=4008)
                return
        except (jwt.ExpiredSignatureError, jwt.InvalidTokenError):
            await websocket.close(code=4008)
            return

        await websocket.accept()
        r: aioredis.Redis = await get_redis()

        # Step 1: Replay buffered lines (handles late joiners — critical for JOB-03)
        buffered = await r.lrange(f"logbuf:{run_id}", 0, -1)
        for line in buffered:
            await websocket.send_text(line.decode("utf-8", errors="replace"))

        # Step 2: Subscribe for live updates
        pubsub = r.pubsub()
        await pubsub.subscribe(f"logs:{run_id}")
        try:
            async for message in pubsub.listen():
                if message["type"] == "message":
                    data = message["data"]
                    text = data.decode("utf-8", errors="replace") if isinstance(data, bytes) else data
                    await websocket.send_text(text)
        except WebSocketDisconnect:
            pass
        finally:
            await pubsub.unsubscribe(f"logs:{run_id}")
            await pubsub.aclose()
    ```

    Token passed as query param `?token=` because WebSocket API does not support custom headers in browsers. This is the standard approach for WS auth.

    **frontend/src/components/LogTerminal/LogTerminal.tsx** — xterm.js terminal:

    Exact scrollback behavior (LOCKED DECISION from CONTEXT.md):
    - `scrollback: 50000` — not 0/unlimited (RESEARCH.md pitfall 7: unlimited causes OOM on long ORFS runs)
    - Auto-scroll enabled by default
    - Pause auto-scroll when user scrolls up (detected via `term.onScroll`)
    - "Jump to bottom" button appears when auto-scroll is paused (fixed position bottom-right of terminal container)
    - Auto-scroll resumes when user scrolls back to the bottom
    - Stage separator lines (starting with "═══") rendered in a distinct color (e.g., `\x1b[36m` cyan) using xterm.js escape code injection

    ```typescript
    import { useEffect, useRef, useState } from "react";
    import { Terminal } from "@xterm/xterm";
    import { FitAddon } from "@xterm/addon-fit";
    import { useLogStream } from "../../hooks/useLogStream";
    import "@xterm/xterm/css/xterm.css";

    interface Props { runId: string | null; }

    export function LogTerminal({ runId }: Props) {
      const termRef = useRef<HTMLDivElement>(null);
      const termInstance = useRef<Terminal | null>(null);
      const [autoScroll, setAutoScroll] = useState(true);
      const [showJumpBtn, setShowJumpBtn] = useState(false);

      // Initialize terminal once
      useEffect(() => {
        const term = new Terminal({
          scrollback: 50000,
          convertEol: true,
          theme: { background: "#0d1117", foreground: "#c9d1d9" },
          fontSize: 13,
          fontFamily: "Menlo, Monaco, 'Courier New', monospace",
        });
        const fit = new FitAddon();
        term.loadAddon(fit);
        if (termRef.current) { term.open(termRef.current); fit.fit(); }

        // Auto-scroll state machine
        term.onScroll(() => {
          const atBottom = term.buffer.active.viewportY >= term.buffer.active.length - term.rows;
          setAutoScroll(atBottom);
          setShowJumpBtn(!atBottom);
        });
        termInstance.current = term;
        return () => { term.dispose(); };
      }, []);

      const handleLine = (line: string) => {
        const t = termInstance.current;
        if (!t) return;
        // Style stage separator lines distinctly
        if (line.startsWith("═══")) {
          t.writeln(`\x1b[36m${line}\x1b[0m`);
        } else {
          t.writeln(line);
        }
        if (autoScroll) t.scrollToBottom();
      };

      useLogStream(runId, handleLine);

      return (
        <div style={{ position: "relative", height: "100%", background: "#0d1117" }}>
          <div ref={termRef} style={{ height: "100%", width: "100%" }} />
          {showJumpBtn && (
            <button onClick={() => { termInstance.current?.scrollToBottom(); setAutoScroll(true); setShowJumpBtn(false); }}
              style={{ position: "absolute", bottom: 16, right: 16 }}>
              Jump to bottom
            </button>
          )}
        </div>
      );
    }
    ```

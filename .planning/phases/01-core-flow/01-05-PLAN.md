    **backend/app/services/metrics_service.py** — Parse ORFS per-stage JSON output files:

    ```python
    import json
    from pathlib import Path

    def parse_ppa_metrics(workspace: str, platform: str, design: str) -> tuple[dict, dict]:
        """
        Parse all per-stage ORFS JSON metric files from logs/{platform}/{design}/base/*.json.

        ORFS writes one JSON file per stage (e.g. 2_1_floorplan.json, 5_1_grt.json,
        6_report.json). Each file uses the key format: {stage}__{category}__{metric}
        Example keys: "floorplan__timing__setup__ws", "route__timing__setup__tns"

        Returns:
            ppa: dict — friendly-name mapped subset for runs.ppa JSONB column
            stage_metrics: dict — raw merged dict of ALL keys for runs.stage_metrics JSONB column

        Reference: CLAUDE.md "Real ORFS Metrics Schema" section
        """
        ppa = {
            "worst_negative_slack": None,
            "total_negative_slack": None,
            "drc_violations": 0,
            "core_area": None,
            "total_power": None,
            "flow_complete": False,
        }
        stage_metrics = {}

        logs_dir = Path(workspace) / "logs" / platform / design / "base"
        if not logs_dir.exists():
            return ppa, stage_metrics

        # Merge all per-stage JSON files in sorted order so later stages override earlier
        for json_file in sorted(logs_dir.glob("*.json")):
            try:
                data = json.loads(json_file.read_text())
                stage_metrics.update(data)
            except (json.JSONDecodeError, OSError):
                pass  # skip malformed files — do not crash

        if not stage_metrics:
            return ppa, stage_metrics

        # Map raw ORFS keys to friendly names for the ppa column.
        # Route metrics preferred over finish for WNS/TNS (finish may not exist for partial runs).
        ppa.update({
            "worst_negative_slack": stage_metrics.get(
                "route__timing__setup__ws",
                stage_metrics.get("finish__timing__setup__ws")
            ),
            "total_negative_slack": stage_metrics.get(
                "route__timing__setup__tns",
                stage_metrics.get("finish__timing__setup__tns")
            ),
            "core_area": stage_metrics.get("floorplan__design__core__area"),
            "drc_violations": int(stage_metrics.get("finish__design__violations", 0)),
            "total_power": stage_metrics.get(
                "finish__power__total",
                stage_metrics.get("route__power__total")
            ),
            # flow_complete: finish stage metrics present indicates full flow ran
            "flow_complete": stage_metrics.get("finish__design__violations") is not None,
        })

        return ppa, stage_metrics

    class MetricsService:
        """Service wrapper for dependency injection."""
        def parse_from_workspace(self, workspace: str, platform: str, design: str) -> tuple[dict, dict]:
            return parse_ppa_metrics(workspace, platform, design)
    ```

    **worker/tasks/tile_generator.py** — KLayout PNG generation (CRITICAL: this is the permanent fast-path — never remove per CLAUDE.md):

    ```python
    from worker.celery_app import app
    import os
    from pathlib import Path

    @app.task(name="worker.tasks.tile_generator.generate_png", queue="background", bind=True)
    def generate_png(self, run_id: str, workspace: str):
        """
        Generate static layout PNG using KLayout Python API (headless, no X11 required).
        This is the fast-path preview. It must remain even after Phase 2 adds tiled viewer.

        PERMANENT FAST PATH — DO NOT REMOVE when Phase 2 tile viewer is added.
        Source: KLayout Python API — https://www.klayout.de/doc-qt5/programming/python.html

        Input: 6_final.gds from results/{platform}/{design}/base/ (written at finish stage).
               DEF (6_final.def) is only written at finish and can be used as fallback.
               Intermediate stage artifacts are .odb files (OpenDB binary) — not suitable
               for KLayout PNG generation directly.

        Also uploads all per-stage .odb files and GDS to MinIO for download links.
        Parses ORFS per-stage JSON metrics and stores in runs.ppa and runs.stage_metrics.
        """
        from app.core.config import get_settings
        settings = get_settings()

        # Find results directory in workspace
        results_dirs = list(Path(workspace).glob("results/*/*/base"))
        if not results_dirs:
            # Try without /base suffix (older ORFS versions)
            results_dirs = list(Path(workspace).glob("results/*/*"))
        if not results_dirs:
            return  # no results yet
        results_dir = results_dirs[0]

        gds_file = results_dir / "6_final.gds"
        def_file = results_dir / "6_final.def"  # only exists at finish stage

        # Upload all per-stage .odb artifacts to MinIO for download links.
        # ORFS intermediate artifact format is .odb (OpenDB binary) — NOT .def.
        # DEF is only written at the finish stage as 6_final.def.
        import boto3
        from botocore.config import Config
        s3 = boto3.client("s3",
            endpoint_url=f"http://{settings.MINIO_ENDPOINT}",
            aws_access_key_id=settings.MINIO_ACCESS_KEY,
            aws_secret_access_key=settings.MINIO_SECRET_KEY,
            config=Config(signature_version="s3v4"),
            region_name="us-east-1")

        odb_stage_map = {
            "1_synth.odb":     "synthesis",
            "2_floorplan.odb": "floorplan",
            "3_place.odb":     "place",
            "4_cts.odb":       "cts",
            "5_route.odb":     "route",
        }
        for odb_filename, stage_name in odb_stage_map.items():
            odb_path = results_dir / odb_filename
            if odb_path.exists():
                with open(odb_path, "rb") as f:
                    s3.put_object(
                        Bucket=settings.S3_BUCKET_ARTIFACTS,
                        Key=f"runs/{run_id}/artifacts/{odb_filename}",
                        Body=f,
                        ContentType="application/octet-stream",
                    )

        # Upload GDS and (if present) DEF
        for artifact_file, key_suffix in [(gds_file, "6_final.gds"), (def_file, "6_final.def")]:
            if artifact_file.exists():
                with open(artifact_file, "rb") as f:
                    s3.put_object(
                        Bucket=settings.S3_BUCKET_ARTIFACTS,
                        Key=f"runs/{run_id}/{key_suffix}",
                        Body=f,
                    )

        # Generate PNG using KLayout Python API (headless batch mode — no display needed)
        import tempfile
        with tempfile.NamedTemporaryFile(suffix=".png", delete=False) as tmp:
            tmp_path = tmp.name

        png_uploaded = False
        try:
            if not gds_file.exists() and not def_file.exists():
                pass  # no layout to render
            else:
                try:
                    import klayout.db as db
                    import klayout.lay as lay
                except ImportError:
                    # KLayout not available in this worker image — log warning, continue
                    # Worker image may need klayout python bindings installed
                    pass
                else:
                    layout = db.Layout()
                    source_path = str(gds_file) if gds_file.exists() else str(def_file)
                    layout.read(source_path)
                    top = layout.top_cell()
                    if top is not None:
                        bbox = top.bbox()
                        view = lay.LayoutView()
                        view.load_layout(layout, True)
                        view.max_hier()
                        view.save_image_with_options(tmp_path, 2048, 2048, 0, 0, 0, bbox)

                        png_key = f"runs/{run_id}/layout.png"
                        with open(tmp_path, "rb") as f:
                            s3.put_object(
                                Bucket=settings.S3_BUCKET_ARTIFACTS,
                                Key=png_key,
                                Body=f,
                                ContentType="image/png",
                            )
                        png_uploaded = True
        finally:
            if os.path.exists(tmp_path):
                os.unlink(tmp_path)

        # Parse ORFS per-stage JSON metrics and update run record.
        # Determine platform and design from directory structure.
        from app.services.metrics_service import parse_ppa_metrics
        rel_parts = results_dir.relative_to(Path(workspace) / "results").parts
        platform = rel_parts[0] if len(rel_parts) >= 1 else "sky130hd"
        design = rel_parts[1] if len(rel_parts) >= 2 else "unknown"
        ppa, stage_metrics = parse_ppa_metrics(workspace, platform, design)

        _update_run_record(run_id, settings, artifact_path=f"runs/{run_id}/", ppa=ppa, stage_metrics=stage_metrics)

    def _update_run_record(run_id: str, settings, artifact_path: str, ppa: dict, stage_metrics: dict):
        """Sync DB update in background task."""
        from sqlalchemy import create_engine, text
        from sqlalchemy.orm import Session
        import json
        sync_url = settings.DATABASE_URL.replace("postgresql+asyncpg://", "postgresql://")
        with Session(create_engine(sync_url)) as db:
            db.execute(text(
                "UPDATE runs SET artifact_path=:ap, ppa=:ppa::jsonb, stage_metrics=:sm::jsonb WHERE id=:id"),
                {"ap": artifact_path, "ppa": json.dumps(ppa), "sm": json.dumps(stage_metrics), "id": run_id})
            db.commit()
    ```

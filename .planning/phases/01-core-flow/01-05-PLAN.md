    ```python
    import json
    from pathlib import Path

    def parse_ppa_metrics(workspace: str, platform: str, design: str) -> dict:
        """
        Primary: logs/{platform}/{design}/metadata.json (METRICS2.1 format)
        Source: https://openroad-flow-scripts.readthedocs.io/en/latest/contrib/Metrics.html
        """
        result = {
            "worst_negative_slack": None,
            "total_negative_slack": None,
            "drc_violations": 0,
            "core_area": None,
            "total_power": None,
            "flow_complete": False,
        }
        metadata_path = Path(workspace) / "logs" / platform / design / "metadata.json"
        if metadata_path.exists():
            try:
                data = json.loads(metadata_path.read_text())
                result.update({
                    "worst_negative_slack": data.get("timing__setup__ws"),
                    "total_negative_slack": data.get("timing__setup__tns"),
                    "drc_violations": int(data.get("route__drc_errors__count", 0)),
                    "core_area": data.get("design__instance__area"),
                    "total_power": data.get("power__total"),
                    "flow_complete": data.get("flow__platform__status") == "succeeded",
                })
            except (json.JSONDecodeError, KeyError):
                pass  # return defaults on parse error — do not crash
        return result

    class MetricsService:
        """Service wrapper for dependency injection."""
        def parse_from_workspace(self, workspace: str, platform: str, design: str) -> dict:
            return parse_ppa_metrics(workspace, platform, design)
    ```

    Note: The ORFS metadata.json key names (`timing__setup__ws`, `route__drc_errors__count`, etc.) are from METRICS2.1 format documented at openroad-flow-scripts.readthedocs.io. Verify these keys against an actual ORFS run during integration testing — they may vary by ORFS version.

    **worker/tasks/tile_generator.py** — KLayout PNG generation (CRITICAL: this is the permanent fast-path — never remove per CLAUDE.md):

    ```python
    from worker.celery_app import app
    import os
    from pathlib import Path

    @app.task(name="worker.tasks.tile_generator.generate_png", queue="background", bind=True)
    def generate_png(self, run_id: str, workspace: str):
        """
        Generate static layout PNG using KLayout Python API (headless, no X11 required).
        This is the fast-path preview. It must remain even after Phase 2 adds tiled viewer.
        Source: KLayout Python API — https://www.klayout.de/doc-qt5/programming/python.html
        """
        from app.core.config import get_settings
        settings = get_settings()

        # Find DEF and LEF files in workspace
        # ORFS output: results/{platform}/{design}/6_final.def and 6_final.gds
        results_dirs = list(Path(workspace).glob("results/*/*"))
        if not results_dirs:
            return  # no results yet
        results_dir = results_dirs[0]
        def_file = results_dir / "6_final.def"
        gds_file = results_dir / "6_final.gds"

        if not def_file.exists() and not gds_file.exists():
            return

        # Generate PNG using KLayout Python API (headless batch mode — no display needed)
        import tempfile
        with tempfile.NamedTemporaryFile(suffix=".png", delete=False) as tmp:
            tmp_path = tmp.name

        try:
            # Import KLayout — if not available in container, log warning and return
            try:
                import klayout.db as db
                import klayout.lay as lay
            except ImportError:
                # KLayout not available in this worker image
                # Log warning: "KLayout not found — layout PNG skipped. Worker image may need klayout installed."
                _update_run_status(run_id, settings, {"artifact_path": f"runs/{run_id}/"})
                return

            layout = db.Layout()
            source_path = str(gds_file) if gds_file.exists() else str(def_file)
            layout.read(source_path)
            top = layout.top_cell()
            if top is None:
                return
            bbox = top.bbox()

            view = lay.LayoutView()
            view.load_layout(layout, True)
            view.max_hier()
            view.save_image_with_options(tmp_path, 2048, 2048, 0, 0, 0, bbox)

            # Upload PNG to MinIO
            import boto3
            from botocore.config import Config
            s3 = boto3.client("s3",
                endpoint_url=f"http://{settings.MINIO_ENDPOINT}",
                aws_access_key_id=settings.MINIO_ACCESS_KEY,
                aws_secret_access_key=settings.MINIO_SECRET_KEY,
                config=Config(signature_version="s3v4"),
                region_name="us-east-1")
            png_key = f"runs/{run_id}/layout.png"
            with open(tmp_path, "rb") as f:
                s3.put_object(Bucket=settings.S3_BUCKET_ARTIFACTS, Key=png_key, Body=f, ContentType="image/png")

            # Also upload GDS and DEF to MinIO for download links
            for artifact_file, key_suffix in [(gds_file, "6_final.gds"), (def_file, "6_final.def")]:
                if artifact_file.exists():
                    with open(artifact_file, "rb") as f:
                        s3.put_object(Bucket=settings.S3_BUCKET_ARTIFACTS,
                                      Key=f"runs/{run_id}/{key_suffix}", Body=f)

            # Update run: set artifact_path, parse PPA metrics, update storage_bytes
            from app.services.metrics_service import parse_ppa_metrics
            # Determine platform and design from directory structure
            platform_dirs = list((Path(workspace) / "results").glob("*"))
            platform = platform_dirs[0].name if platform_dirs else "sky130hd"
            design_dirs = list(platform_dirs[0].glob("*")) if platform_dirs else []
            design = design_dirs[0].name if design_dirs else "unknown"
            ppa = parse_ppa_metrics(workspace, platform, design)

            _update_run_record(run_id, settings, artifact_path=f"runs/{run_id}/", ppa=ppa)

        finally:
            os.unlink(tmp_path, missing_ok=True)

    def _update_run_record(run_id: str, settings, artifact_path: str, ppa: dict):
        """Sync DB update in background task."""
        from sqlalchemy import create_engine, text
        from sqlalchemy.orm import Session
        import json
        sync_url = settings.DATABASE_URL.replace("postgresql+asyncpg://", "postgresql://")
        with Session(create_engine(sync_url)) as db:
            db.execute(text(
                "UPDATE runs SET artifact_path=:ap, ppa=:ppa::jsonb WHERE id=:id"),
                {"ap": artifact_path, "ppa": json.dumps(ppa), "id": run_id})
            db.commit()
    ```

    **worker/container/manager.py** — Docker SDK wrapper:
    ```python
    import docker
    from docker.errors import NotFound, APIError

    class ContainerManager:
        def __init__(self):
            self._client = docker.from_env()

        def run_container(self, run_id: str, image: str, workspace_path: str,
                          pdk_root: str, settings: dict) -> "Container":
            return self._client.containers.run(
                image=image,
                command=["make", f"DESIGN_CONFIG=/workspace/config.mk"],
                name=f"orfs_job_{run_id}",
                detach=True,
                network_mode="none",                    # CRITICAL: no network access
                cpu_period=100000,
                cpu_quota=settings["JOB_CPU_CORES"] * 100000,
                mem_limit=f"{settings['JOB_RAM_GB']}g",
                memswap_limit=f"{settings['JOB_RAM_GB']}g",  # no swap
                read_only=True,
                tmpfs={"/tmp": "size=512m"},
                user="orfs:orfs",
                cap_drop=["ALL"],
                security_opt=["no-new-privileges"],
                volumes={
                    workspace_path: {"bind": "/workspace", "mode": "rw"},
                    pdk_root: {"bind": "/pdks", "mode": "ro"},
                },
                # storage_opt size requires overlay2 + pquota — try, fallback if unsupported
                # storage_opt={"size": f"{settings['JOB_DISK_GB']}G"},
            )

        def stop_and_remove(self, container_or_name, timeout: int = 10) -> None:
            try:
                c = self._client.containers.get(container_or_name) if isinstance(container_or_name, str) else container_or_name
                c.stop(timeout=timeout)
                c.remove(force=True)
            except NotFound:
                pass  # already gone — this is fine

        def list_orfs_containers(self) -> list[dict]:
            containers = self._client.containers.list(filters={"name": "orfs_job_"})
            return [{"name": c.name, "run_id": c.name.replace("orfs_job_", ""),
                     "status": c.status, "id": c.id} for c in containers]
    ```

    Note on storage-opt: Commented out by default because `storage_opt size=` requires `overlay2` with `pquota` mount option on RHEL/Rocky 9 (confirmed pitfall from RESEARCH.md). Include a commented block with instructions for enabling it. Use filesystem quotas at the OS level as the alternative.

    **worker/tasks/orfs_job.py** — Full ORFS task:
    ```python
    from worker.celery_app import app
    from worker.container.manager import ContainerManager
    import redis, json, re, os, asyncio
    from datetime import datetime, timezone

    STAGE_PATTERNS = {
        "synthesis":  re.compile(r"(Starting|Finished)\s+synthesis", re.IGNORECASE),
        "floorplan":  re.compile(r"(Starting|Finished)\s+floorplan", re.IGNORECASE),
        "place":      re.compile(r"(Starting|Finished)\s+placement", re.IGNORECASE),
        "cts":        re.compile(r"(Starting|Finished)\s+cts", re.IGNORECASE),
        "route":      re.compile(r"(Starting|Finished)\s+routing", re.IGNORECASE),
        "gds":        re.compile(r"(Starting|Finished)\s+final", re.IGNORECASE),
    }
    SEPARATOR_FMT = "═══ {stage} ══════════════════════════════════"

    @app.task(name="worker.tasks.orfs_job.run_orfs_job", queue="orfs_jobs", bind=True)
    def run_orfs_job(self, run_id: str):
        from app.core.config import get_settings
        settings = get_settings()
        r = redis.Redis.from_url(settings.REDIS_URL)
        # Use sync DB access in Celery task (not asyncpg)
        from sqlalchemy import create_engine, text
        from sqlalchemy.orm import Session
        sync_db_url = settings.DATABASE_URL.replace("postgresql+asyncpg://", "postgresql://")
        engine = create_engine(sync_db_url)

        container = None
        manager = ContainerManager()

        def publish_line(line: str):
            channel = f"logs:{run_id}"
            list_key = f"logbuf:{run_id}"
            r.publish(channel, line)
            r.rpush(list_key, line)
            r.ltrim(list_key, -5000, -1)   # keep last 5000 lines
            r.expire(list_key, 86400)       # 24hr TTL

        def update_status(status: str, stage: str = None):
            with Session(engine) as db:
                db.execute(text(
                    "UPDATE runs SET status=:s, stage_completed=COALESCE(:st, stage_completed), "
                    "completed_at=CASE WHEN :s IN ('complete','failed','cancelled','timeout') "
                    "THEN NOW() ELSE completed_at END WHERE id=:id"),
                    {"s": status, "st": stage, "id": run_id})
                db.commit()

        try:
            # Prepare workspace: download source files from MinIO to /tmp/workspace_{run_id}/
            workspace = f"/tmp/workspace_{run_id}"
            os.makedirs(workspace, exist_ok=True)
            # Download project files from MinIO using storage_service
            # (simplified: actual MinIO download via boto3 — fetch artifact_path from DB)
            update_status("starting")
            publish_line(f"[chipatelier] Starting ORFS job {run_id}")

            container = manager.run_container(run_id, settings.ORFS_IMAGE, workspace,
                                              settings.PDK_ROOT, settings.__dict__)
            update_status("running")

            # Stream logs
            for raw_line in container.logs(stream=True, follow=True):
                line = raw_line.decode("utf-8", errors="replace").rstrip()
                # Stage transition detection
                for stage, pattern in STAGE_PATTERNS.items():
                    if pattern.search(line):
                        separator = SEPARATOR_FMT.format(stage=stage.upper())
                        publish_line(separator)
                        update_status("running", stage)
                        break
                publish_line(line)

            result = container.wait(timeout=settings.JOB_TIMEOUT_SECONDS)
            exit_code = result.get("StatusCode", 1)

            # Trigger artifact collection (plan 01-05 implements the task body)
            final_status = "complete" if exit_code == 0 else "failed"
            update_status(final_status)
            publish_line(f"[chipatelier] Job {run_id} finished with status: {final_status}")

            if exit_code == 0:
                # Dispatch background tasks for PNG generation and metrics parsing
                from worker.tasks.tile_generator import generate_png
                generate_png.delay(run_id, workspace)

        except Exception as exc:
            update_status("failed")
            publish_line(f"[chipatelier] Job {run_id} failed with exception: {exc}")
            raise
        finally:
            if container:
                manager.stop_and_remove(container)
            # Clean up workspace
            import shutil
            shutil.rmtree(workspace, ignore_errors=True)
    ```

    **worker/tasks/watchdog.py** — Celery beat task for orphaned container cleanup:
    ```python
    @app.task(name="worker.tasks.watchdog.cleanup_orphaned_containers", queue="background")
    def cleanup_orphaned_containers():
        """Run every 2 minutes via Celery beat. Finds ORFS containers with no active run in DB."""
        manager = ContainerManager()
        containers = manager.list_orfs_containers()
        for c in containers:
            run_id = c["run_id"]
            # Check DB: if run not in running/starting state, stop container
            with Session(engine) as db:
                row = db.execute(text("SELECT status FROM runs WHERE id=:id"), {"id": run_id}).first()
                if row is None or row.status not in ("queued", "starting", "running"):
                    manager.stop_and_remove(c["name"])
    ```
